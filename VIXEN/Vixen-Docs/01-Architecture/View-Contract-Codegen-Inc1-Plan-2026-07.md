```csharp
        public static IReadOnlyList<INamedTypeSymbol> LoadViews(IEnumerable<string> schemaFiles)
        {
            var trees = schemaFiles.Select(f => CSharpSyntaxTree.ParseText(File.ReadAllText(f), path: f));
            var comp = CSharpCompilation.Create("viewschemas", trees, BuildRefs());
            var result = new List<INamedTypeSymbol>();
            foreach (var tree in comp.SyntaxTrees)
            {
                var model = comp.GetSemanticModel(tree);
                foreach (var node in tree.GetRoot().DescendantNodes()
                             .Where(n => n is StructDeclarationSyntax || n is ClassDeclarationSyntax))
                {
                    if (model.GetDeclaredSymbol(node) is INamedTypeSymbol sym &&
                        sym.GetAttributes().Any(a => a.AttributeClass != null && a.AttributeClass.Name == "ViewAttribute"))
                        result.Add(sym);
                }
            }
            return result;
        }
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using Microsoft.CodeAnalysis;

namespace Yeroket.KernelFramework.Codegen
{
    public enum ViewScalar { Int, Float, Bool, String }
    public enum ViewFieldKind { Scalar, Struct, StructArray }

    public sealed class ViewField
    {
        public string Name { get; }
        public ViewFieldKind Kind { get; }
        public ViewScalar? Scalar { get; }
        public ViewStruct Struct { get; }    // element type for Struct / StructArray
        public ViewField(string name, ViewFieldKind kind, ViewScalar? scalar, ViewStruct str)
        { Name = name; Kind = kind; Scalar = scalar; Struct = str; }
    }

    public sealed class ViewStruct
    {
        public string Name { get; }
        public IReadOnlyList<ViewField> Fields { get; }
        public ViewStruct(string name, IReadOnlyList<ViewField> fields) { Name = name; Fields = fields; }
    }

    public static class ViewModelBuilder
    {
        public static ViewStruct Build(INamedTypeSymbol viewType)
        {
            var fields = new List<ViewField>();
            foreach (var f in viewType.GetMembers().OfType<IFieldSymbol>()
                         .Where(f => !f.IsStatic && !f.IsConst && !f.IsImplicitlyDeclared))
            {
                fields.Add(Classify(f));
            }
            return new ViewStruct(viewType.Name, fields);
        }

        static ViewField Classify(IFieldSymbol f)
        {
            var s = TryScalar(f.Type);
            if (s.HasValue) return new ViewField(f.Name, ViewFieldKind.Scalar, s.Value, null);

            // T[] where T is a struct → StructArray
            if (f.Type is IArrayTypeSymbol arr && arr.ElementType is INamedTypeSymbol elem && IsPlainStruct(elem))
                return new ViewField(f.Name, ViewFieldKind.StructArray, null, Build(elem));

            // plain struct field → Struct
            if (f.Type is INamedTypeSymbol named && IsPlainStruct(named))
                return new ViewField(f.Name, ViewFieldKind.Struct, null, Build(named));

            throw new NotSupportedException($"View field '{f.Name}' has unsupported type '{f.Type.ToDisplayString()}' (only int/float/bool/string scalars, struct fields, and struct[] arrays are allowed).");
        }

        static ViewScalar? TryScalar(ITypeSymbol t)
        {
            switch (t.SpecialType)
            {
                case SpecialType.System_Int32:   return ViewScalar.Int;
                case SpecialType.System_Single:  return ViewScalar.Float;
                case SpecialType.System_Boolean: return ViewScalar.Bool;
                case SpecialType.System_String:  return ViewScalar.String;
                default: return null;
            }
        }

        // A "plain struct" here = a value/reference type declared in the schema (has fields), NOT a builtin.
        static bool IsPlainStruct(INamedTypeSymbol t) =>
            (t.TypeKind == TypeKind.Struct || t.TypeKind == TypeKind.Class) &&
            t.SpecialType == SpecialType.None &&
            t.GetMembers().OfType<IFieldSymbol>().Any(m => !m.IsStatic && !m.IsImplicitlyDeclared);
    }
}
```

```csharp
using System.Collections.Generic;
using System.Linq;
using System.Text;

namespace Yeroket.KernelFramework.Codegen
{
    public static class RmlDataModelEmitter
    {
        public static string Emit(ViewStruct v)
        {
            var sb = new StringBuilder();
            sb.AppendLine("#pragma once");
            sb.AppendLine("// GENERATED by Yeroket kernel-codegen (--view) — DO NOT EDIT. Regenerate from the canonical [View] schema.");
            sb.AppendLine("#include <RmlUi/Core/DataModelHandle.h>");
            sb.AppendLine("#include <vector>");
            sb.AppendLine("#include <string>");
            sb.AppendLine();
            sb.AppendLine("namespace Vixen::Views {");
            sb.AppendLine();

            // Row structs = the element types of Struct / StructArray fields (dedup, depth-first).
            var rows = new List<ViewStruct>();
            CollectStructs(v, rows, new HashSet<string>());
            foreach (var r in rows) EmitRowStruct(sb, r);

            // Bound-pointer bundle (host owns the storage; passed in so the fn is standalone-compilable).
            sb.AppendLine($"struct {v.Name}Bind {{");
            foreach (var f in v.Fields) sb.AppendLine($"    {BindPtrType(f)} {f.Name};");
            sb.AppendLine("};");
            sb.AppendLine();

            // The registration function.
            sb.AppendLine($"inline void Bind{v.Name}Model(Rml::DataModelConstructor& c, const {v.Name}Bind& b) {{");
            // 1. RegisterStruct + members (row structs, in the order they appear as fields)
            foreach (var r in rows)
            {
                sb.AppendLine($"    if (auto sh = c.RegisterStruct<{r.Name}>()) {{");
                foreach (var rf in r.Fields)
                    sb.AppendLine($"        sh.RegisterMember(\"{rf.Name}\", &{r.Name}::{rf.Name});");
                sb.AppendLine("    }");
            }
            // 2. RegisterArray for each StructArray field
            foreach (var f in v.Fields.Where(f => f.Kind == ViewFieldKind.StructArray))
                sb.AppendLine($"    c.RegisterArray<std::vector<{f.Struct.Name}>>();");
            // 3. Bind each field (scalars + arrays), in field order
            foreach (var f in v.Fields)
                sb.AppendLine($"    c.Bind(\"{f.Name}\", b.{f.Name});");
            sb.AppendLine("}");
            sb.AppendLine();
            sb.AppendLine("}  // namespace Vixen::Views");
            return sb.ToString().Replace("\r\n", "\n");
        }

        static void CollectStructs(ViewStruct v, List<ViewStruct> outList, HashSet<string> seen)
        {
            foreach (var f in v.Fields)
            {
                if ((f.Kind == ViewFieldKind.Struct || f.Kind == ViewFieldKind.StructArray) && f.Struct != null && seen.Add(f.Struct.Name))
                {
                    CollectStructs(f.Struct, outList, seen);   // nested-first
                    outList.Add(f.Struct);
                }
            }
        }

        static void EmitRowStruct(StringBuilder sb, ViewStruct r)
        {
            sb.Append($"struct {r.Name} {{ ");
            foreach (var f in r.Fields) sb.Append($"{CppScalar(f.Scalar!.Value)} {f.Name}; ");
            sb.AppendLine("};");
        }

        static string CppScalar(ViewScalar s) => s switch
        {
            ViewScalar.Int => "int",
            ViewScalar.Float => "float",
            ViewScalar.Bool => "bool",
            ViewScalar.String => "Rml::String",
            _ => "int",
        };

        static string BindPtrType(ViewField f) => f.Kind switch
        {
            ViewFieldKind.Scalar => CppScalar(f.Scalar!.Value) + "*",
            ViewFieldKind.Struct => f.Struct.Name + "*",
            ViewFieldKind.StructArray => $"std::vector<{f.Struct.Name}>*",
            _ => "void*",
        };
    }
}
```
