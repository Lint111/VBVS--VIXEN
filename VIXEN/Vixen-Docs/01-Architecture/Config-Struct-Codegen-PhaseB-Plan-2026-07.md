```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using Microsoft.CodeAnalysis;

namespace Yeroket.KernelFramework.Codegen
{
    public enum ScalarKind { U32, I32, F32 }
    public enum FieldClass { Scalar, Float3, Mat4, Nested, Array }

    public static class ScalarKindExt
    {
        public static int SizeBytes(this ScalarKind k) => 4;
        public static int AlignBytes(this ScalarKind k) => 4;
    }

    public sealed class ArrayInfo
    {
        public FieldLayout Element { get; }
        public int Count { get; }
        public int Stride { get; }
        public ArrayInfo(FieldLayout element, int count, int stride) { Element = element; Count = count; Stride = stride; }
    }

    public sealed class FieldLayout
    {
        public string Name { get; }
        public FieldClass Class { get; }
        public ScalarKind? Scalar { get; }
        public StructModel Nested { get; }
        public ArrayInfo Array { get; }
        public int Offset { get; set; }   // assigned by StructLayout.Build after alignment
        public int Size { get; }
        public int Align { get; }
        public FieldLayout(string name, FieldClass cls, ScalarKind? scalar, StructModel nested, ArrayInfo array, int offset, int size, int align)
        { Name = name; Class = cls; Scalar = scalar; Nested = nested; Array = array; Offset = offset; Size = size; Align = align; }
        public ScalarKind Kind => Scalar ?? throw new InvalidOperationException($"Field '{Name}' is not a scalar");
    }

    public sealed class StructModel
    {
        public string Name { get; }
        public IReadOnlyList<FieldLayout> Fields { get; }
        public int SizeBytes { get; }
        public StructModel(string name, IReadOnlyList<FieldLayout> fields, int sizeBytes) { Name = name; Fields = fields; SizeBytes = sizeBytes; }
    }

    public static class StructLayout
    {
        public static StructModel Build(INamedTypeSymbol structType, Func<INamedTypeSymbol, StructModel> resolveNested = null)
        {
            resolveNested = resolveNested ?? (nested => Build(nested));
            var fields = new List<FieldLayout>();
            int cursor = 0;
            int structAlign = 4;
            foreach (var f in structType.GetMembers().OfType<IFieldSymbol>()
                         .Where(f => !f.IsStatic && !f.IsConst && !f.IsImplicitlyDeclared))
            {
                var raw = ClassifyField(f, resolveNested);
                cursor = RoundUp(cursor, raw.Align);
                raw.Offset = cursor;
                structAlign = Math.Max(structAlign, raw.Align);
                fields.Add(raw);
                cursor += raw.Size;
            }
            return new StructModel(structType.Name, fields, RoundUp(cursor, structAlign));
        }

        static FieldLayout ClassifyField(IFieldSymbol f, Func<INamedTypeSymbol, StructModel> resolveNested)
        {
            if (IsMarkerType(f.Type, "Float3")) return new FieldLayout(f.Name, FieldClass.Float3, null, null, null, 0, 12, 16);
            if (IsMarkerType(f.Type, "Mat4"))   return new FieldLayout(f.Name, FieldClass.Mat4,   null, null, null, 0, 64, 16);

            var scalar = TryMapScalar(f.Type);
            if (scalar.HasValue)
            {
                var n = GpuArrayLength(f);
                if (n.HasValue)
                {
                    var elem = new FieldLayout(f.Name + "[0]", FieldClass.Scalar, scalar, null, null, 0, 4, 4);
                    int stride = RoundUp(elem.Size, elem.Align);
                    return new FieldLayout(f.Name, FieldClass.Array, null, null, new ArrayInfo(elem, n.Value, stride), 0, stride * n.Value, elem.Align);
                }
                return new FieldLayout(f.Name, FieldClass.Scalar, scalar, null, null, 0, 4, 4);
            }

            if (f.Type is INamedTypeSymbol named && IsGpuStruct(named))
            {
                var nestedModel = resolveNested(named);
                int nestedAlign = nestedModel.Fields.Count > 0 ? nestedModel.Fields.Max(fl => fl.Align) : 4;
                var n = GpuArrayLength(f);
                if (n.HasValue)
                {
                    var elem = new FieldLayout(f.Name + "[0]", FieldClass.Nested, null, nestedModel, null, 0, nestedModel.SizeBytes, nestedAlign);
                    int stride = RoundUp(elem.Size, elem.Align);
                    return new FieldLayout(f.Name, FieldClass.Array, null, null, new ArrayInfo(elem, n.Value, stride), 0, stride * n.Value, nestedAlign);
                }
                return new FieldLayout(f.Name, FieldClass.Nested, null, nestedModel, null, 0, nestedModel.SizeBytes, nestedAlign);
            }

            throw new NotSupportedException($"Unsupported field type '{f.Type.ToDisplayString()}' for field '{f.Name}'");
        }

        static bool IsMarkerType(ITypeSymbol t, string name) =>
            t.Name == name && t.ContainingNamespace != null &&
            t.ContainingNamespace.ToDisplayString() == "Yeroket.Util.KernelFramework";

        static bool IsGpuStruct(INamedTypeSymbol t) =>
            t.GetAttributes().Any(a => a.AttributeClass != null && a.AttributeClass.Name == "GpuStructAttribute");

        static int? GpuArrayLength(IFieldSymbol f)
        {
            var attr = f.GetAttributes().FirstOrDefault(a => a.AttributeClass != null && a.AttributeClass.Name == "GpuArrayAttribute");
            return attr == null ? (int?)null : (int)attr.ConstructorArguments[0].Value;
        }

        static ScalarKind? TryMapScalar(ITypeSymbol t)
        {
            switch (t.SpecialType)
            {
                case SpecialType.System_UInt32: return ScalarKind.U32;
                case SpecialType.System_Int32:  return ScalarKind.I32;
                case SpecialType.System_Single: return ScalarKind.F32;
                default: return null;
            }
        }

        static int RoundUp(int v, int a) => (v + a - 1) / a * a;
    }
}
```

```csharp
using System.Collections.Generic;
using System.IO;
using System.Linq;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;

namespace Yeroket.KernelFramework.Codegen
{
    public static class CompilationLoader
    {
        public static IReadOnlyList<INamedTypeSymbol> LoadGpuStructs(IEnumerable<string> schemaFiles)
        {
            var trees = schemaFiles.Select(f => CSharpSyntaxTree.ParseText(File.ReadAllText(f), path: f));
            var runtimeDir = Path.GetDirectoryName(typeof(object).Assembly.Location);
            var refs = new[]
            {
                MetadataReference.CreateFromFile(typeof(object).Assembly.Location),
                MetadataReference.CreateFromFile(Path.Combine(runtimeDir, "System.Runtime.dll")),
                MetadataReference.CreateFromFile(typeof(Yeroket.Util.KernelFramework.GpuStructAttribute).Assembly.Location),
            };
            var comp = CSharpCompilation.Create("schemas", trees, refs);
            var result = new List<INamedTypeSymbol>();
            foreach (var tree in comp.SyntaxTrees)
            {
                var model = comp.GetSemanticModel(tree);
                foreach (var node in tree.GetRoot().DescendantNodes().OfType<StructDeclarationSyntax>())
                {
                    if (model.GetDeclaredSymbol(node) is INamedTypeSymbol sym &&
                        sym.GetAttributes().Any(a => a.AttributeClass != null && a.AttributeClass.Name == "GpuStructAttribute"))
                        result.Add(sym);
                }
            }
            return result;
        }
    }
}
```
